```c
is_valid_addr (const void *addr) 
{
  return addr != NULL && is_user_vaddr (addr) && pagedir_get_page (thread_current ()->pagedir, addr) != NULL;

static bool
is_all_valid_addr (const void *addr, size_t size) 
{
  const char *byte = addr;
  for (size_t i = 0; i < size; i++) {
    if (!is_valid_addr (byte + i)) {
      return false;
    }
  }
  return true;
}

static bool
is_valid_string (const char *str)
{
  if (!is_valid_addr (str))
    return false;
  
  size_t len = 0;
  while (true) {
    if (!is_valid_addr (str + len))
      return false;
    if (str[len] == '\0')
      break;
    len++;
  }
  return true;
}
```

```c
static int
syscall_read (int fd, void *buffer, unsigned size)
{
  if (!is_valid_addr (buffer))
    syscall_exit (-1);

  if (fd == STDIN_FILENO)
  {
    int bytes_read = 0;
    uint8_t *buf = buffer;
    for (unsigned i = 0; i < size; i++)
    {
      int c = input_getc ();
      if (c == -1)
        break;
      if (!is_valid_addr (buf + i))
        syscall_exit (-1);
      buf[i] = (uint8_t) c;
      bytes_read++;
    }
    return bytes_read;
  }

  struct thread *cur = thread_current ();
  struct process *proc = cur->process;

  lock_acquire (&proc->lock);
  if (fd < 0 || fd >= MAX_FD_COUNT || proc->fd_table[fd] == NULL)
  {
    lock_release (&proc->lock);
    return -1;
  }
  struct file *f = proc->fd_table[fd];
  lock_release (&proc->lock);

  lock_acquire (&filesys_lock);
  int bytes_read = file_read (f, buffer, size);
  lock_release (&filesys_lock);
  return bytes_read;
}

static int
syscall_write (int fd, const void *buffer, unsigned size)
{
  if (!is_all_valid_addr (buffer, size))
    syscall_exit (-1);
    
  if (fd == STDOUT_FILENO)
  {
    putbuf (buffer, size);
    return size;
  }

  struct thread *cur = thread_current ();
  struct process *proc = cur->process;

  lock_acquire (&proc->lock);
  if (fd < 0 || fd >= MAX_FD_COUNT || proc->fd_table[fd] == NULL)
  {
    lock_release (&proc->lock);
    return -1;
  }
  struct file *f = proc->fd_table[fd];
  lock_release (&proc->lock);
```

```c
tid_t
process_execute (const char *cmd_line) 
{
  char **argv;
  int argc = 0;
  argv = palloc_get_page (0);
  if (argv == NULL)
    return TID_ERROR; // 这里没有成功分配任何东西
  char *cmd_line_copy = palloc_get_page (0);
  if (cmd_line_copy == NULL)
    {
      palloc_free_page (argv); // 这里只成功分配了 argv
      return TID_ERROR;
    }
  strlcpy (cmd_line_copy, cmd_line, PGSIZE);
  char *save_ptr;
  for (char *token = strtok_r (cmd_line_copy, " ", &save_ptr); token != NULL;
       token = strtok_r (NULL, " ", &save_ptr))
    {
      if ((size_t) argc >= PGSIZE / sizeof(char *))
        {
          palloc_free_page (argv);
          palloc_free_page (cmd_line_copy); // 这里分配了 argv 和 cmd_line_copy
          return TID_ERROR;
        }
      argv[argc++] = token;
    }
  argv[argc] = NULL;

  struct thread *cur = thread_current ();
  struct process *proc = cur->process;
  struct process *child_proc = malloc (sizeof (struct process));
  if (child_proc == NULL)
    {
      palloc_free_page (argv);
      palloc_free_page (cmd_line_copy); // 这里分配了 argv 和 cmd_line_copy，没有成功产生子进程
      return TID_ERROR;
    }
  init_process (child_proc, argv);
  child_proc->cmd_line_cpy = cmd_line_copy;
  tid_t tid = thread_create (argv[0], PRI_DEFAULT, start_process, child_proc);
  if (tid == TID_ERROR)
    {
      process_refcount_free (child_proc); // 这里 argv 和 cmd_line_copy 全部转移给了子进程，由子进程负责释放
      return TID_ERROR;
    }
  child_proc->pid = tid;
  lock_acquire (&proc->lock);
  list_push_back (&proc->children, &child_proc->elem);
  lock_release (&proc->lock);

  sema_down (&child_proc->wait_sema); // Wait for the child process to load.
  if (!child_proc->loaded) // If loading failed, return -1.
    {
      lock_acquire (&proc->lock);
      list_remove (&child_proc->elem);
      lock_release (&proc->lock);
      process_refcount_free (child_proc); // 同上
      return TID_ERROR;
    }
  return tid;
}
```

>B8: Consider parent process P with child process C.  How do you
>ensure proper synchronization and avoid race conditions when P
>calls wait(C) before C exits?  After C exits?  How do you ensure
>that all resources are freed in each case?  How about when P
>terminates without waiting, before C exits?  After C exits?  Are
>there any special cases?

`struct process` 里面存有信号量和布尔变量 `waited`。

- P 在 C 退出之前调用 `wait`：查找到子进程 C，设置 `waited` 为 `true`，修改信号量使 P 进入睡眠阻塞，直到 C 退出唤醒父进程；
- P 在 C 退出之后调用 `wait`：由于信号量已经被修改，所以 `sema_down` 不会让 P 进入阻塞。

资源释放：

C 自己退出时会设置 `exit_status` 和 `exited`，同时修改信号量用于唤醒可能存在的等待者，并使用 `process_refcount_free` 减少 C 自己的引用计数。

P 成功等待后会将 C 从子进程列表中删除，并再次使用 `process_refcount_free` 减少 C 的引用计数。由于引用计数初始为 2，此时 C 会完全释放其已有资源。

如果父进程 P 未等待就终止，那么 C 就成为了“僵尸进程”，`parent` 指针不再有效；然而我们并不会主动使用 `parent`，所以不会崩溃。然而 C 会永远泄露，这也是没有办法的事情啊，毕竟我们没有 init 进程；或者我可以写一个孤儿标记，之后有机会再写。

>B10: What advantages or disadvantages can you see to your design
>for file descriptors?

每个进程的 `struct process` 中包含一个固定大小的数组 `struct file *fd_table[MAX_FD_COUNT]`。

文件描述符是整数索引，从 2 开始分配（0 和 1 预留给标准输入和标准输出）。

打开文件时，遍历 `fd_table` 找到第一个空闲位置，将 `struct file *` 存入，并返回该索引。

关闭文件时，将对应槽位清空，并调用 `file_close`。

文件描述符仅在进程内部有效，不同进程可以有相同的 fd 编号。

主要是写着简单功能完整。问题就是每次都要线性查找。
